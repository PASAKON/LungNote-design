CSS Tokens Reference

Copy these CSS variables into your project. All skeleton/loader components use these tokens.

/* ── LungNote Skeleton Tokens ── */

/* Light mode */
:root {
  --skel-base: #e8e5de;
  --skel-shine: #f5f3ee;
}

/* Dark mode */
[data-theme="dark"] {
  --skel-base: #333330;
  --skel-shine: #3e3e3a;
}

/* ── Animations ── */
@keyframes shimmer {
  0% { background-position: -400px 0; }
  100% { background-position: 400px 0; }
}
@keyframes pulse {
  0%, 100% { opacity: 1; }
  50% { opacity: 0.4; }
}
@keyframes spin {
  to { transform: rotate(360deg); }
}
@keyframes dots {
  0%, 80%, 100% { transform: scale(0); }
  40% { transform: scale(1); }
}
@keyframes bounce {
  0%, 80%, 100% { transform: scaleY(0.4); }
  40% { transform: scaleY(1); }
}

/* ── Base skeleton ── */
.skel {
  background: var(--skel-base);
  border-radius: 4px;
  position: relative;
  overflow: hidden;
}
.skel.shimmer::after {
  content: '';
  position: absolute;
  inset: 0;
  background: linear-gradient(
    90deg, transparent 0%,
    var(--skel-shine) 50%,
    transparent 100%
  );
  background-size: 400px 100%;
  animation: shimmer 1.5s ease-in-out infinite;
}
.skel.pulse {
  animation: pulse 1.5s ease-in-out infinite;
}

/* ── Spinner ── */
.spinner-ring {
  width: 40px; height: 40px;
  border: 3px solid var(--border);
  border-top-color: var(--accent);
  border-radius: 50%;
  animation: spin 0.8s linear infinite;
}
.spinner-ring.sm { width: 24px; height: 24px; border-width: 2.5px; }
.spinner-ring.lg { width: 56px; height: 56px; border-width: 4px; }
.spinner-ring.xl { width: 72px; height: 72px; border-width: 5px; }

/* ── Dots ── */
.spinner-dots { display: flex; gap: 8px; }
.spinner-dots .dot {
  width: 12px; height: 12px;
  border-radius: 50%;
  background: var(--accent);
  animation: dots 1.4s ease-in-out infinite;
}
.spinner-dots .dot:nth-child(2) { animation-delay: .16s; }
.spinner-dots .dot:nth-child(3) { animation-delay: .32s; }
